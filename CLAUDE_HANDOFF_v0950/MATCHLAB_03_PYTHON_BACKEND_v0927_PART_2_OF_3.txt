"""Source-verified club history extraction: honours, roster snapshots, imagery and links.

Only parse Wikipedia page *after* its Wikidata identity was matched by the caller.
This module never claims a Wikipedia snapshot is an up-to-the-minute roster.
A missing statistic is missing, not zero. No club title totals are inferred.
"""
import datetime as dt
import html
import re
import unicodedata
import urllib.parse
from pathlib import Path

_HEADING = re.compile(r'(?m)^\s*(={2,6})\s*([^\n=]+?)\s*\1\s*$')
_HONOURS = re.compile(r'^(honou?rs|titles|trophies|achievements|palmar[eè]s|t[ií]tulos|erfolge|başarılar|kupalar|trofeylər|nailiyyətlər)$', re.I)
_SQUAD = re.compile(r'^(current squad|first[- ]team squad|first[- ]team players|senior squad|playing squad|squad|plantilla actual|primer equipo|kad(r|ro)|heyət|əsas heyət)$', re.I)


def _norm(text):
    text = str(text or '').casefold().replace('ə','e').replace('ı','i')
    return ''.join(c for c in unicodedata.normalize('NFKD',text) if not unicodedata.combining(c))


def _text(text):
    """Strip basic wiki markup safely; skip templates rather than inventing content."""
    text=str(text or '')
    text=re.sub(r'<!--.*?-->','',text,flags=re.S)
    text=re.sub(r'<ref\b[^>]*>.*?</ref\s*>|<ref\b[^>]*/>', '',text,flags=re.S|re.I)
    text=re.sub(r'<[^>]*>','',text)
    text=re.sub(r'\[\[(?:[^\]|]+\|)?([^\]]+)\]\]',r'\1',text)
    text=re.sub(r'\{\{[^{}]*\}\}','',text)
    text=re.sub(r"'{2,5}",'',text)
    return html.unescape(re.sub(r'\s+',' ',text)).strip(' \t:;*|')[:220]


def section(wikitext, heading_pattern):
    """Return text bounded by next header at the same or higher level."""
    if not isinstance(wikitext,str):return ''
    matches=list(_HEADING.finditer(wikitext))
    for i,m in enumerate(matches):
        name=_text(m.group(2))
        if heading_pattern.fullmatch(_norm(name)):
            level=len(m.group(1))
            end=next((n.start() for n in matches[i+1:] if len(n.group(1))<=level),len(wikitext))
            return wikitext[m.end():min(end,m.end()+90_000)]
    return ''


def _table_honours(section_text):
    """Parse Type/Competition/Titles/Seasons wiki tables when counts are explicit.

    Most major football-club articles use tabular honours, not bullet lists.
    This deliberately ignores totals that cannot be attributed to a single cup.
    """
    out=[];group=''
    for chunk in re.split(r'(?m)^\s*\|-.*$',section_text)[1:]:
        if chunk.strip().startswith('!') or chunk.strip().startswith('\n!'):
            continue
        cells=[]
        for line in chunk.splitlines():
            line=line.strip()
            if line.startswith('|}'):
                break
            if not line.startswith('|') or line.startswith('|-'):
                continue
            val=line[1:].strip()
            # Handle '| rowspan="5" | Domestic' and similar MediaWiki cell attrs.
            if re.match(r'(?i)^(?:rowspan|colspan|style|scope|class)\s*=',val):
                val=val.split('|',1)[-1].strip() if '|' in val else ''
            for item in re.split(r'\s*\|\|\s*',val):
                item=_text(item)
                if item:cells.append(item)
        if len(cells)<2:continue
        categories={'domestic','continental','intercontinental','international','worldwide','regional','national','other'}
        if _norm(cells[0]).strip() in categories and len(cells)>=3:
            group=cells[0]
            title,count=cells[1],cells[2]
        else:
            title,count=cells[0],cells[1]
        count_str=re.sub(r'(?i)\s*\[[a-z]+\]\s*$','',count).strip()
        # Some Wikipedia cells carry footnote superscripts after the numeral.
        match=re.match(r'^(\d{1,3})(?:\D.*)?$',count_str)
        if not match:continue
        number=int(match.group(1))
        if number<1 or number>150 or not 4<=len(title)<=105:continue
        if _norm(title).strip() in categories or 'runner' in _norm(title):continue
        out.append({'name':title,'count':number,'category':group or 'Kuboklar','source':'Wikipedia · Honours (cədvəl)'})
        if len(out)>=50:break
    return out


def parse_honours(wikitext):
    """Read only explicitly listed wins. Counts require an explicit source numeral."""
    raw=section(wikitext,_HONOURS)
    if not raw:return []
    entries=[];group='';seen=set()
    for line in raw.splitlines():
        heading=_HEADING.fullmatch(line.strip())
        if heading:
            group=_text(heading.group(2))[:70]
            continue
        if not re.match(r'^\s*\*{1,3}\s+',line):continue
        item=_text(re.sub(r'^\s*\*{1,3}\s+','',line)).strip()
        if not item or len(item)<4:continue
        if re.search(r'(?i)runner[s]?[- ]?up|finalists?|semi[- ]final|third place|second place|runners? up',item):continue
        if re.search(r'(?i)\b(reserve[s]?|youth|under[- ]?\d\d|academy|women)\b',group+' '+item):continue
        # Typical Wikipedia: "UEFA Champions League: Winners (15): years", "La Liga (36)".
        main=item.split(':')[0].strip()
        if re.search(r'(?i)\b(winners?|champions?|title[s]?)\b',main):
            # if competition is after a prefix like "Winners", keep context only if explicit.
            main=re.sub(r'(?i)\s*[-–:]?\s*(?:winners?|champions?)\s*$', '',main).strip()
        number=re.search(r'(?i)\b(?:winners?|champions?|titles?)\s*\(\s*(\d{1,3})\s*\)',item)
        if not number:number=re.search(r'\(\s*(\d{1,3})\s*\)\s*(?::|$)',main)
        if not number:number=re.search(r'(?i)^(.*?):\s*(\d{1,3})\b',item)
        count=int(number.group(number.lastindex)) if number and 0<int(number.group(number.lastindex))<=150 else None
        name=re.sub(r'\s*\(\s*\d{1,3}\s*\)\s*$','',main)
        name=re.sub(r'(?i)\s*[—–-]?\s*(?:winners?|champions?)\s*$','',name)
        name=name.strip(' :—–-')
        if not 4<=len(name)<=105:continue
        if re.fullmatch(r'(?i)domestic|international|continental|regional|national|other|friendly|major|minor|worldwide',name):continue
        # A trophy's source may have count but no competition: do not invent.
        if name.casefold() in ('winners','champions','runners-up'):continue
        key=_norm(name)
        if key in seen:continue
        seen.add(key)
        entries.append({'name':name,'count':count,'category':group or 'Kuboklar','source':'Wikipedia · Honours'})
        if len(entries)>=45:break
    for h in _table_honours(raw):
        key=_norm(h['name'])
        if key not in seen:
            seen.add(key)
            entries.append(h)
    return entries[:50]


def _templates(raw,names=('fs player','football squad player','football squad member')):
    """Yield balanced templates with a recognized opening name."""
    pattern=re.compile(r'\{\{\s*('+('|'.join(re.escape(n).replace('\\ ','\\s+') for n in names))+r')\b',re.I)
    for match in pattern.finditer(raw):
        pos=match.start();depth=0;i=pos
        while i<len(raw) and i<pos+5000:
            token=raw[i:i+2]
            if token=='{{':depth+=1;i+=2;continue
            if token=='}}':
                depth-=1;i+=2
                if depth==0:
                    yield raw[pos:i];break
                continue
            i+=1


def parse_squad(wikitext):
    """Wikipedia squad snapshot, never a certified live squad."""
    raw=section(wikitext,_SQUAD)
    if not raw:return []
    from wikipedia_enrichment import split_params
    players=[];names=set()
    for item in _templates(raw):
        params=split_params(item)
        name=_text(params.get('name') or params.get('player') or '')
        number=_text(params.get('no') or params.get('number') or '')
        role=_text(params.get('pos') or params.get('position') or '')
        if not 2<len(name)<85 or name.casefold() in ('tbd','vacant','n/a'):continue
        if not re.fullmatch(r'\d{1,2}',number):number=''
        if role and len(role)>12:role=''
        key=_norm(name)
        if key in names:continue
        names.add(key)
        players.append({'name':name,'number':number,'position':role,'source':'Wikipedia · Current squad'})
        if len(players)>=40:break
    return players


def colors_from_text(raw):
    """Only a visual swatch when the source explicitly gives its hex color."""
    raw=str(raw or '')
    hexes=list(dict.fromkeys('#'+h.upper() for h in re.findall(r'(?<![0-9A-Fa-f])#([0-9A-Fa-f]{6})(?![0-9A-Fa-f])',raw)))
    return hexes[:5]


def stadium_media(club_entity, get_json, max_filename=200):
    """P115 -> stadium P18 -> Commons imageinfo and license, all linked by QIDs."""
    from wikipedia_enrichment import entity_value
    stadium=entity_value(club_entity,'P115')
    qid=stadium.get('id','') if isinstance(stadium,dict) else ''
    if not re.fullmatch(r'Q[1-9][0-9]{0,11}',qid):return {}
    url='https://www.wikidata.org/w/api.php?'+urllib.parse.urlencode({'action':'wbgetentities','format':'json','ids':qid,'props':'labels|claims','languages':'az|tr|en','languagefallback':'1'})
    payload=(get_json(url,86400).get('entities') or {}).get(qid,{})
    name=next((payload.get('labels',{}).get(lang,{}).get('value') for lang in ('az','tr','en') if payload.get('labels',{}).get(lang,{}).get('value')),'')
    photo=entity_value(payload,'P18')
    if not isinstance(photo,str) or not 1<len(photo)<max_filename:return {}
    file='File:'+photo.replace('_',' ')
    params={'action':'query','format':'json','formatversion':2,'prop':'imageinfo','iiprop':'url|extmetadata','iiurlwidth':720,'titles':file}
    media_url='https://commons.wikimedia.org/w/api.php?'+urllib.parse.urlencode(params)
    pages=(get_json(media_url,86400).get('query') or {}).get('pages') or []
    if not pages or len(pages)!=1:return {}
    info=(pages[0].get('imageinfo') or [{}])[0]
    ext=info.get('extmetadata') or {}
    def val(key):return str((ext.get(key) or {}).get('value') or '')
    if val('NonFree').casefold()=='true':return {}
    license_name=_text(val('LicenseShortName'))
    license_url=val('LicenseUrl')
    # Reject unlicensed files rather than claiming they are free.
    if not license_name or not license_url.startswith('https://'):return {}
    if not re.search(r'(?i)\bcc[ -]?(?:by|zero|0)|creative commons|public domain|\bpd\b|\bgfdl\b',license_name):return {}
    image=info.get('thumburl') or info.get('url') or ''
    if not isinstance(image,str) or not image.startswith('https://upload.wikimedia.org/'):return {}
    page=info.get('descriptionurl') or 'https://commons.wikimedia.org/wiki/'+urllib.parse.quote(file.replace(' ','_'),safe=':_')
    if not page.startswith('https://commons.wikimedia.org/'):return {}
    return {'stadiumPhoto':image,'stadiumPhotoPage':page,'stadiumPhotoLicense':license_name,'stadiumPhotoLicenseUrl':license_url,'stadiumPhotoCredit':_text(val('Attribution') or val('Artist'))[:160],'stadiumQid':qid,'stadiumImageName':name}


def historical_links(club_name, seed, team_qid='', aliases=()):
    """Curated archival associations only. No name-based cross-person unification."""
    from wikipedia_enrichment import VALID_QID
    keys={_norm(x).strip() for x in (club_name,*aliases) if x}

"""Free, source-labeled football fixture/table adapters for MATCHLAB AZ.

OpenFootball football.json is public-domain historical data (CC0). It is *not*
a real-time score feed. football-data.org's free tier needs a separately
registered token, exposes delayed scores and limits requests to 10/minute.
No credentials are stored in the repository or returned through the API.
"""
import datetime as dt
import json
import re
import threading
import time
import urllib.error
import urllib.parse
import urllib.request

OPENFOOTBALL_CODES={
    'eng.1':'en.1','ger.1':'de.1','esp.1':'es.1','ita.1':'it.1',
    'fra.1':'fr.1','ned.1':'nl.1','por.1':'pt.1','tur.1':'tr.1',
}
FOOTBALL_DATA_CODES={
    'eng.1':'PL','ger.1':'BL1','esp.1':'PD','ita.1':'SA','fra.1':'FL1',
    'ned.1':'DED','por.1':'PPL','uefa.champions':'CL',
}
LOCK=threading.Lock()
_REQUESTS=[]
_CACHE={}


def openfootball_url(league,date):
    if league not in OPENFOOTBALL_CODES:return ''
    day=dt.date.fromisoformat(date)
    # The public-domain archive is a historical fallback, not a live source.
    if day>dt.date.today()-dt.timedelta(days=2) or day.year<1990:
        return ''
    year=day.year if day.month>=7 else day.year-1
    folder=f'{year}-{str((year+1)%100).zfill(2)}'
    return f'https://raw.githubusercontent.com/openfootball/football.json/master/{folder}/{OPENFOOTBALL_CODES[league]}.json'


def _score(value):
    if isinstance(value,bool):return None
    if isinstance(value,int) and 0<=value<=60:return value
    return None


def parse_openfootball(data,league,date):
    if not isinstance(data,dict) or not isinstance(data.get('matches'),list):
        raise ValueError('OpenFootball formatı gözlənilən kimi deyil')
    rows=[]
    for i,match in enumerate(data['matches'][:2000]):
        if not isinstance(match,dict) or match.get('date')!=date:continue
        home=str(match.get('team1') or '').strip()[:110]
        away=str(match.get('team2') or '').strip()[:110]
        if not home or not away:continue
        ft=(match.get('score') or {}).get('ft') if isinstance(match.get('score'),dict) else None
        hs,as_=(ft[0],ft[1]) if isinstance(ft,list) and len(ft)==2 else (None,None)
        hs,as_=_score(hs),_score(as_)
        final=hs is not None and as_ is not None
        rows.append({
            'id':f'openfootball:{league}:{date}:{i}',
            'eventId':'', 'league':league,
            'home':{'id':'','name':home,'image':''},
            'away':{'id':'','name':away,'image':''},
            'homeScore':hs if final else None,'awayScore':as_ if final else None,
            'state':'final' if final else 'scheduled',
            'date':date+'T12:00:00Z','minute':'',
            'kickoffUnverified':True,
            'source':'OpenFootball · CC0 tarixi arxiv (canlı deyil)',
        })
        if len(rows)>=70:break
    return rows


def football_data_path(league,date=None,table=False):
    code=FOOTBALL_DATA_CODES.get(league)
    if not code:return ''
    if table:return f'/v4/competitions/{code}/standings'
    dt.date.fromisoformat(date)
    return f'/v4/competitions/{code}/matches?'+urllib.parse.urlencode({'dateFrom':date,'dateTo':date})


def _num(x):
    return int(x) if type(x) is int and 0<=x<=100000 else None


def parse_football_data_matches(data,league,date):
    if not isinstance(data,dict) or not isinstance(data.get('matches'),list):
        raise ValueError('football-data.org formatı gözlənilən kimi deyil')
    rows=[]
    for match in data['matches'][:100]:
        home=match.get('homeTeam') or {};away=match.get('awayTeam') or {}
        when=match.get('utcDate') or ''
        if not (isinstance(when,str) and when.startswith(date)):
            continue
        if not (home.get('name') and away.get('name')):continue
        status=str(match.get('status') or '').upper()
        # The free plan is delayed; NEVER advertise this as real-time.
        state='final' if status=='FINISHED' else 'scheduled'
        score=(match.get('score') or {}).get('fullTime') or {}
        hs,as_=_num(score.get('home')),_num(score.get('away'))
        rows.append({
            'id':'football-data:'+str(match.get('id') or ''),'eventId':str(match.get('id') or ''),
            'league':league,
            'home':{'id':str(home.get('id') or ''),'name':home['name'],'image':''},
            'away':{'id':str(away.get('id') or ''),'name':away['name'],'image':''},
            'homeScore':hs if state=='final' else None,'awayScore':as_ if state=='final' else None,
            'state':state,'date':when,'minute':'',
            'source':'football-data.org · pulsuz gecikmiş məlumat',
        })
    return rows


def parse_football_data_table(data):
    standings=data.get('standings') if isinstance(data,dict) else None
    if not isinstance(standings,list):raise ValueError('football-data.org cədvəli yanlışdır')
    total=next((x for x in standings if x.get('type')=='TOTAL' and isinstance(x.get('table'),list)),None)
    if not total:return []
    rows=[]
    for item in total['table'][:30]:
        club=item.get('team') or {}
        if not club.get('name'):continue
        rows.append({
            'name':str(club['name'])[:120],'image':'',
            'rank':_num(item.get('position')),
            'played':_num(item.get('playedGames')),
            'win':_num(item.get('won')),
            'draw':_num(item.get('draw')),
            'loss':_num(item.get('lost')),
            'points':_num(item.get('points')),
            'gf':_num(item.get('goalsFor')),'ga':_num(item.get('goalsAgainst')),
        })
    return rows


def fetch_football_data(path,token,opener=None):
    """Safe allowlisted official API with free-tier throttling and cached JSON."""
    if not re.fullmatch(r'/v4/competitions/(?:PL|BL1|PD|SA|FL1|DED|PPL|CL)/(?:standings|matches\?dateFrom=\d{4}-\d{2}-\d{2}&dateTo=\d{4}-\d{2}-\d{2})',path):
        raise ValueError('Yalnız təsdiqlənmiş sorğular icazəlidir')
    if not token or not isinstance(token,str) or len(token)>250 or any(ord(c)<33 or ord(c)>126 for c in token):
        raise ValueError('football-data.org pulsuz tokeni konfiqurasiya edilməyib')
    if opener is None:opener=urllib.request.urlopen
    now=time.monotonic()
    key=(path,token)
    with LOCK:
        cached=_CACHE.get(key)
        if cached and cached[0]>now:return cached[1]
        _REQUESTS[:]=[t for t in _REQUESTS if now-t<60]
        if len(_REQUESTS)>=9:raise ConnectionError('football-data.org pulsuz 10 sorğu/dəqiqə limitinə yaxındır')
        _REQUESTS.append(now)
    req=urllib.request.Request('https://api.football-data.org'+path,headers={
        'X-Auth-Token':token,'Accept':'application/json','User-Agent':'MATCHLAB-AZ/0.9.23'})
    with opener(req,timeout=8) as response:
        if int(response.headers.get('Content-Length','0') or 0)>2_000_000:
            raise ValueError('Böyük mənbə cavabı')
        data=response.read(2_000_001)
        if len(data)>2_000_000:raise ValueError('Böyük mənbə cavabı')
    result=json.loads(data)
    with LOCK:
        _CACHE[key]=(time.monotonic()+300,result)
        if len(_CACHE)>100:
            for k in list(_CACHE)[:len(_CACHE)-100]:_CACHE.pop(k,None)
    return result

"""Source-labeled match details for free match providers.

No synthetic lineups, stats, game events, or claimed formations. Supports
ESPN (unofficial API), TheSportsDB v1 free, and community OpenLigaDB. Parsers
are pure functions suitable for offline fixture-based tests.
"""
from __future__ import annotations

import re


def short(value, limit=140):
    return str(value or '').strip()[:limit]


def is_yes(value):
    return value is True or str(value).strip().lower() in ('yes','true','1')


def numeric(value):
    if value is None or str(value).strip() == '':
        return None
    try:
        num = float(str(value).replace(',', '').replace('%', ''))
        if num != num or abs(num) > 10000000:
            return None
        return int(num) if num.is_integer() else round(num, 2)
    except (TypeError, ValueError, OverflowError):
        return None


def safe_formation(value):
    name = short(value, 24).strip()

"""MATCHLAB AZ v0.9.27 — public HTTPS hosting adapter (Python stdlib).

Deployment uses the existing validated database and API, but blocks local maintenance,
dangerous upload/install routes, and unnecessary user-triggered upstream refreshes.
"""
import os, json, time, threading, urllib.parse
from http.server import ThreadingHTTPServer
from server import Handler as LocalHandler, send_data
import world_db

RATE_LIMIT = 180
RATE_WINDOW = 60.0
RATE_LOCK = threading.Lock()
RATE_BUCKETS = {}

def is_allowed(ip):
    now = time.monotonic()
    with RATE_LOCK:
        if len(RATE_BUCKETS) > 5000:
            for key, (ts, amount) in list(RATE_BUCKETS.items()):
                if now - ts > RATE_WINDOW: RATE_BUCKETS.pop(key, None)
        ts, amount = RATE_BUCKETS.get(ip, (now, 0))
        if now - ts > RATE_WINDOW: ts, amount = now, 0
        amount += 1
        RATE_BUCKETS[ip] = ts, amount
        return amount <= RATE_LIMIT

class PublicHandler(LocalHandler):
    server_version = 'MATCHLAB/0.9.27'

    def do_POST(self):
        # Critical: Internet visitors must not start large imports or upload arbitrary files.
        self.json_response(403, {'error': 'Onlayn versiyada fayl yükləmə və indeks yeniləmə bağlıdır.'})

    def do_GET(self):
        path = urllib.parse.urlsplit(self.path).path
        if path == '/healthz':
            reg=world_db.counts()
            ok=bool(reg and reg.get('total') == 506207)
            return self.json_response(200 if ok else 503, {'ok': ok, 'service': 'matchlab', 'version': '0.9.27', 'registryTotal':reg.get('total',0) if reg else 0, 'hosting':'render'})
        if path in ('/api/install-start', '/api/install-upload', '/api/install-status', '/api/wikipedia-refresh'):
            return self.json_response(403, {'error': 'İctimai serverdə bu əməliyyat aktiv deyil.'})
        if path == '/api/status':
            try:
                result = send_data(path,{})
                result.update({'version': '0.9.27', 'publicHosting': True, 'canInstall': False})
                return self.json_response(200, result)
            except Exception:
                return self.json_response(503, {'ok': False, 'error': 'Baza əlçatan deyil'})
        # Only same-origin requests are needed; no CORS for third party sites.
        xf = self.headers.get('X-Forwarded-For', '')
        # The rightmost forwarded hop is less vulnerable to forged leading XFF values.
        ip = xf.split(',')[-1].strip()[:70] if xf else self.client_address[0]
        if not is_allowed(ip):
            return self.json_response(429, {'error': 'Sorğu limiti: bir az sonra yenidən yoxla.'})
        if path.startswith('/api/') and len(self.path) > 1200:
            return self.json_response(414, {'error': 'Sorğu çox uzundur'})
        return super().do_GET()

def main():
    port = int(os.environ.get('PORT', '10000'))
    if not (1 <= port <= 65535): raise ValueError('Invalid PORT')
    with ThreadingHTTPServer(('0.0.0.0', port), PublicHandler) as httpd:
        print(f'MATCHLAB AZ onlayn server: 0.0.0.0:{port}', flush=True)
        httpd.serve_forever(poll_interval=0.5)

if __name__ == '__main__': main()

def _normal(value):
    raw = str(value or '').casefold().replace('ə','e').replace('ı','i')
    raw = ''.join(c for c in unicodedata.normalize('NFKD', raw) if not unicodedata.combining(c))
    return ' '.join(re.findall(r'[^\W_]+', raw))


def qid_of(row):
    if not isinstance(row, dict):
        return ''
    identifier = str(row.get('id') or '')
    mapped = VERIFIED_LEGACY_QIDS.get(identifier)
    qid = mapped or row.get('wikidataId') or row.get('qid') or (identifier if _QID.fullmatch(identifier) else '')
    return str(qid) if _QID.fullmatch(str(qid)) else ''


def roles_of(row):
    roles = []
    for role in [*(row.get('roles') or []), row.get('kind'), *VERIFIED_ROLES.get(qid_of(row), ())]:
        if role in KINDS and role not in roles:
            roles.append(role)
    return roles


def decorate(row):
    r = dict(row)
    qid = qid_of(r)
    if qid and r.get('kind') in KINDS:
        r['qid'] = qid
        r['roles'] = roles_of(r)
    return r


def _priority(row):
    identifier = str(row.get('id') or '')
    return (3 if _REGISTRY.fullmatch(identifier) else 2 if _QID.fullmatch(identifier) else 1 if identifier in VERIFIED_LEGACY_QIDS else 0)


def unify(rows):
    """One result per verified QID, preserving all separate verified persons."""
    output = []
    by_key = {}
    for item in rows:
        r = decorate(item)
        qid = qid_of(r)
        key = 'person:' + qid if qid and r.get('kind') in KINDS else 'id:' + str(r.get('id') or '')
        if key not in by_key:
            by_key[key] = len(output)
            output.append(r)
            continue
        pos = by_key[key]
        old = output[pos]
        roles = list(dict.fromkeys(roles_of(old) + roles_of(r)))
        if _priority(r) > _priority(old):
            old = r
        old['roles'] = roles
        if qid:
            old['qid'] = qid
        output[pos] = old
    return output


def suppress_reviewed_display_duplicates(rows):
    """Hide unlinked, full-name, country-matched legacy duplicates in search only.

    No inference on unknown names, no collapsing different QIDs. Source rows
    remain intact and can still be opened via their exact IDs.
    """
    linked = {}
    for row in rows:
        if row.get('kind') not in KINDS:
            continue
        key = (_normal(row.get('name')), _normal(row.get('country')))
        qid = qid_of(row)
        if qid:
            linked.setdefault(key, set()).add(qid)
    out = []
    for row in rows:
        if row.get('kind') not in KINDS or qid_of(row):
            out.append(row)
            continue
        key = (_normal(row.get('name')), _normal(row.get('country')))
        target = REVIEWED_DISPLAY_NAMES.get(key)
        if target and linked.get(key) == {target}:
            continue
        out.append(row)
    return out

"""MATCHLAB AZ live scoreboard aggregation.

Data is always upstream supplied (never guessed). The 'all' view fetches all
configured competitions concurrently with bounded work and short-lived caches.
A checkedAt timestamp refers to OUR last upstream check, not proof of provider
real-time latency. ESPN endpoints are undocumented and best-effort.
"""
import concurrent.futures
import datetime
import threading
import time
import league_catalog
from zoneinfo import ZoneInfo

LEAGUE_NAMES = {
    'eng.1': 'Premier League', 'esp.1': 'La Liga', 'ger.1': 'Bundesliga',
    'ita.1': 'Serie A', 'fra.1': 'Ligue 1', 'uefa.champions': 'Champions League',
    'uefa.europa': 'Europa League', 'tur.1': 'Süper Lig',
    'por.1': 'Primeira Liga', 'ned.1': 'Eredivisie',
    'usa.1': 'MLS', 'ksa.1': 'Saudi Pro League',
}
LEAGUE_NAMES.update(league_catalog.EXTRA_LEAGUES)
_LOCK = threading.Lock()
_CACHE = {}
_FLIGHT = {}


def baku_today():
    return datetime.datetime.now(ZoneInfo('Asia/Baku')).date().isoformat()


def _compute_view(league, date, load_one):
    """Produce honest match lists. load_one(league, date) is injected for tests."""
    if league != 'all' and league not in LEAGUE_NAMES:
        return {'items': [], 'available': False, 'message': 'Liqa mövcud deyil'}
    ttl = 25 if date == baku_today() else 600
    key = (league, date)
    now = time.monotonic()
    with _LOCK:
        old = _CACHE.get(key)
        if old and now - old[0] < ttl:
            return dict(old[1], cached=True)
    targets = list(league_catalog.CORE_LEAGUES) if league == 'all' else [league]
    results = {}
    def load(name):
        try:
            r = load_one(name, date)
            if not isinstance(r, dict) or not isinstance(r.get('items'), list):
                raise ValueError('unexpected provider response')
            return name, r
        except Exception as exc:
            return name, {'items': [], 'available': False, 'partial': True,
                          'source': name, 'message': type(exc).__name__}
    if league == 'all':
        # 6 concurrent adapters bound per-user latency; the API cache prevents
        # every open browser tab from hammering the sources.
        with concurrent.futures.ThreadPoolExecutor(max_workers=6) as executor:
            for name, result in executor.map(load, targets):
                results[name] = result
    else:
        name, result = load(league)
        results[name] = result
    all_items = []
    errors = []
    seen = set()
    partial = False
    ok = 0
    for name in targets:
        r = results[name]
        if r.get('available'):
            ok += 1
        else:
            errors.append(name)
        partial = partial or bool(r.get('partial')) or not bool(r.get('available')) or bool(r.get('stale'))
        for item in r['items']:
            if not isinstance(item, dict):
                continue
            item = dict(item)
            item['league'] = name
            item['leagueName'] = LEAGUE_NAMES[name]
            item['stale'] = bool(r.get('stale'))
            ident = str(item.get('id') or '')
            if not ident or ident in seen:
                continue
            seen.add(ident)
            all_items.append(item)
    def sort_key(match):
        # A genuine live match is placed first, then fixtures by kick-off time.
        priority = {'live': 0, 'scheduled': 1, 'final': 2}.get(match.get('state'), 3)
        return (priority, str(match.get('date') or ''), match.get('leagueName', ''))
    all_items.sort(key=sort_key)
    result = {
        'items': all_items, 'available': ok > 0,
        'partial': partial,
        'stale': not bool(ok) and bool(all_items),
        'source': f'{ok}/{len(targets)} liqa mənbəsi' if league == 'all' else results[league].get('source', ''),
        'message': ('Bəzi liqalar üçün canlı məlumat əlçatan deyil: '+', '.join(errors)) if errors else '',
        'checkedAt': datetime.datetime.now(datetime.timezone.utc).isoformat(timespec='seconds'),
        'coverage': {'checked': len(targets), 'available': ok, 'failedLeagues': errors},
        'liveCount': sum(1 for x in all_items if x.get('state') == 'live' and not x.get('stale')),
    }
    if not ok and not all_items:
        result['message'] = 'Seçilmiş liqalar üçün mənbə cavab vermədi.'
    with _LOCK:
        _CACHE[key] = (time.monotonic(), result)
        # Bound the cache to 40 date/league keys.
        if len(_CACHE) > 40:
            for k in list(_CACHE)[:len(_CACHE)-40]:
                _CACHE.pop(k, None)
    return result


def view(league, date, load_one):
    """Coalesce concurrent tabs for a single date/league to one upstream check."""
    if league != 'all' and league not in LEAGUE_NAMES:
        return {'items': [], 'available': False, 'message': 'Liqa mövcud deyil'}
    key = (league, date)
    ttl = 25 if date == baku_today() else 600
    with _LOCK:
        old = _CACHE.get(key)
        if old and time.monotonic() - old[0] < ttl:
            return dict(old[1], cached=True)
        lock = _FLIGHT.setdefault(key, threading.Lock())
    with lock:
        with _LOCK:
            old = _CACHE.get(key)
            if old and time.monotonic() - old[0] < ttl:
                return dict(old[1], cached=True)
        return _compute_view(league, date, load_one)

"""MATCHLAB AZ verified Wikimedia media expansion (v0.9.17).

Second-stage lookup when Wikidata P18 (person) or P154 (club emblem) is empty.
Never identify by a similar-looking name: resolve the QID's own Wikipedia
sitelink and require the article's pageprops.wikibase_item to match the QID.
Use only MediaWiki pageimages marked free; a club's stadium photo is not a logo.
"""
import datetime as dt
import json
import re
import sqlite3
import threading
import urllib.parse
from contextlib import closing
from pathlib import Path

VALID_ID = re.compile(r'^Q[1-9]\d{0,11}$')
LANGUAGES = ('en', 'tr', 'az', 'de', 'es', 'fr', 'it', 'pt')
KINDS = frozenset(('players', 'coaches', 'teams'))
CACHE_FILE = Path(__file__).with_name('wikimedia_media_cache.sqlite')
DB_LOCK = threading.Lock()
# Deliberately strict. A club photograph, stadium and jersey are NOT crests.
NOT_PORTRAIT = re.compile(r'(?:^|[ _-])(logo|emblem|badge|crest|stadium|ground|jersey|shirt|flag|coat.of.arms|map|kit)(?:[ _.-]|$)', re.I)
CREST_WORD = re.compile(r'(^|[^a-z])(logo|logotipo|logotype|badge|crest|emblem|escudo|wappen|vereinswappen|insignia|brasao|brasão|coat.of.arms|shield|grb|znak)([^a-z]|$)', re.I)


def commons_file_page(filename):
    if not isinstance(filename,str) or not filename or len(filename)>245:return ''
    return 'https://commons.wikimedia.org/wiki/File:'+urllib.parse.quote(filename.removeprefix('File:').replace(' ', '_'),safe='()_')


def free_image_url(value):
    """Allow only real Wikimedia upload CDN files, not attacker-controlled hosts."""
    if not isinstance(value,str) or not value.startswith('https://upload.wikimedia.org/') or len(value)>1500:return ''
    parts=urllib.parse.urlsplit(value)
    if parts.hostname!='upload.wikimedia.org' or parts.username or parts.password:return ''
    if not parts.path.startswith('/wikipedia/'):return ''
    return value


def crest_like(filename):
    if not isinstance(filename,str):return False
    text=urllib.parse.unquote(filename.replace('_',' ')).casefold()
    # SVG illustrations often include flags or stadium plans. Require explicit logo words.
    return bool(CREST_WORD.search(text))


def _connect():
    con=sqlite3.connect(CACHE_FILE,timeout=2)
    con.execute('CREATE TABLE IF NOT EXISTS media (qid TEXT NOT NULL, kind TEXT NOT NULL, stamp TEXT NOT NULL, payload TEXT NOT NULL, PRIMARY KEY(qid,kind))')
    return con


def cache_load(qid,kind):
    if not VALID_ID.fullmatch(qid or '') or kind not in KINDS:return None
    try:
        with DB_LOCK:
            with closing(_connect()) as con, con:
                record=con.execute('SELECT stamp,payload FROM media WHERE qid=? AND kind=?',(qid,kind)).fetchone()
        if not record:return None
        data=json.loads(record[1]); age=dt.datetime.now(dt.timezone.utc)-dt.datetime.fromisoformat(record[0])
        # Recheck unavailable entries daily; confirmed media every 3 weeks.
        ttl=dt.timedelta(days=21 if data.get('url') else 1)
        return data if dt.timedelta(0)<=age<ttl else None
    except (sqlite3.Error, OSError, ValueError, TypeError):return None


def cache_write(qid,kind,value):
    if not VALID_ID.fullmatch(qid or '') or kind not in KINDS:return
    try:
        with DB_LOCK:
            with closing(_connect()) as con, con:
                con.execute('INSERT OR REPLACE INTO media VALUES(?,?,?,?)',(qid,kind,dt.datetime.now(dt.timezone.utc).isoformat(),json.dumps(value,ensure_ascii=False)))
    except (sqlite3.Error, OSError, ValueError, TypeError):pass


def cache_stats():
    try:
        with DB_LOCK:
            with closing(_connect()) as con, con:
                entries=con.execute('SELECT kind,payload FROM media').fetchall()
        result={kind:{'found':0,'unavailable':0} for kind in KINDS}
        for kind,payload in entries:
            if kind in result:
                try:result[kind]['found' if json.loads(payload).get('url') else 'unavailable']+=1
                except (ValueError,TypeError):pass
        return result
    except (sqlite3.Error, OSError):return {kind:{'found':0,'unavailable':0} for kind in KINDS}


def verified_page_images(qids,kind,get_json,refresh=False):

"""MATCHLAB AZ: conservative, request-limited football season statistics.

TheSportsDB V1 public API: searches are on demand, and the free player-stats
endpoint can return at most 10 statistic records per player. Never infer that
missing stats are zero, or that the first search hit is the right person.
A full date-of-birth plus a compatible name is required for linkage (unless
an externally verified provider ID has been explicitly recorded).

Provider terms (2026): free API is for development projects; app-store
publication requires a paid subscription. No site scraping.
"""
import datetime as dt
import re
import unicodedata
import urllib.parse

ROOT = 'https://www.thesportsdb.com/api/v1/json/123/'


def normalize(value):
    raw = str(value or '').casefold().replace('ə', 'e').replace('ı', 'i').replace('ł', 'l')
    raw = ''.join(c for c in unicodedata.normalize('NFKD', raw) if not unicodedata.combining(c))
    return ' '.join(re.findall(r'[^\W_]+', raw, re.UNICODE))


def iso_birth(value):
    """Accept unambiguous *full* dates. Year-only is insufficient for identity."""
    value = str(value or '').strip()
    m = re.fullmatch(r'(\d{4})-(\d{2})-(\d{2})(?:T[^\s]*)?', value)
    if not m:
        return ''
    try:
        return dt.date(*map(int, m.groups())).isoformat()
    except ValueError:
        return ''


def name_compatible(profile_name, provider_name):
    a = normalize(profile_name)
    b = normalize(provider_name)
    if not a or not b:
        return False
    if a == b:
        return True
    # Two distinctive shared tokens + identical full DOB. Handles
    # "Lionel Messi" vs "Lionel Andrés Messi" without fuzzy conflation.
    aa = {x for x in a.split() if len(x) >= 3}
    bb = {x for x in b.split() if len(x) >= 3}
    return len(aa & bb) >= 2



def _country(value):
    """Normalize nationality adjectives without accepting a vague country guess."""
    known={
        'spain':'spain','spanish':'spain','italy':'italy','italian':'italy',
        'england':'england','english':'england','france':'france','french':'france',
        'germany':'germany','german':'germany','portugal':'portugal','portuguese':'portugal',
        'brazil':'brazil','brazilian':'brazil','argentina':'argentina','argentinian':'argentina',
        'netherlands':'netherlands','dutch':'netherlands','belgium':'belgium','belgian':'belgium',
        'turkey':'turkey','turkish':'turkey','azerbaijan':'azerbaijan','azerbaijani':'azerbaijan',
    }
    cleaned=normalize(value)
    return known.get(cleaned,cleaned)


def _team_compatible(left,right):
    def tokens(value):
        return {x for x in normalize(value).split() if x not in {'fc','cf','sc','afc','football','club','fk','ac','the','de'} and len(x)>2}
    a,b=tokens(left),tokens(right)
    return bool(a and b and a==b)


def verified_provider_id(profile, results):
    """Only safe namesake linkages: full DOB or a stricter independent check.

    In cases with unknown DOB, require an EXACT full name, same nonempty
    nationality and a historically documented club matching the provider's
    current club. A sole name match must never be enough.
    """
    birth=iso_birth(profile.get('birth'))
    if profile.get('kind') not in ('players','coaches'):
        return None, 'Bu profil futbolçu/məşqçi kimliyi deyil.'
    candidates=[]
    if not birth:
        if not _country(profile.get('country')):
            return None, 'Doğum tarixi və etibarlı ölkə məlumatı yoxdur.'
        clubs=[c if isinstance(c,str) else c.get('name','') for c in (profile.get('clubs') or [])]
        clubs+=list(profile.get('localHistoricalClubs') or [])
        clubs=[x for x in clubs if isinstance(x,str) and x.strip()]
        if not clubs:
            return None, 'Tam doğum tarixi yoxdur, əlavə klub əlaqəsi ilə kimlik təsdiqlənmədi.'
    else:clubs=[]
    for x in results if isinstance(results,list) else []:
        if not isinstance(x,dict) or normalize(x.get('strSport')) not in ('soccer','football'):
            continue
        ident=str(x.get('idPlayer') or '')
        if not re.fullmatch(r'\d{1,12}',ident):continue
        if birth:
            if iso_birth(x.get('dateBorn') or x.get('strDateBorn') or x.get('dateBirth') or '')!=birth:
                continue
            if not name_compatible(profile.get('name'),x.get('strPlayer')):continue
        else:
            if normalize(profile.get('name'))!=normalize(x.get('strPlayer')):
                continue
            nat=_country(x.get('strNationality') or x.get('strCountry'))
            if not nat or nat!=_country(profile.get('country')):continue
            team=x.get('strTeam') or ''
            if not any(_team_compatible(team,old) for old in clubs):continue
        candidates.append(ident)
    ids=list(dict.fromkeys(candidates))
    if len(ids)==1:return ids[0],''
    if len(ids)>1:return None,'Bir neçə uyğun şəxs tapıldı; yanlış assist göstərilmir.'
    if birth:return None,'TheSportsDB-də tam doğum tarixi və ada uyğun futbolçu tapılmadı.'
    return None,'TheSportsDB-də ad, ölkə və təsdiqli klub ilə uyğun şəxs tapılmadı.'


def stat_category(value):
    x = normalize(value)
    if x in ('assists','assist','goal assists','goal assist','total assists','total goal assists'):
        return 'assists'
    if x in ('goals','goal','goals scored','total goals','scored goals'):
        return 'goals'
    if x in ('appearances','appearance','matches played','games played','matches','games','total appearances'):
        return 'appearances'
    if x in ('minutes played','minutes','playing time minutes'):
        return 'minutes'
    if x in ('yellow cards','yellow card'):
        return 'yellowCards'
    if x in ('red cards','red card'):
        return 'redCards'
    return None


def valid_stat(value):
    if isinstance(value, bool):
        return None
    s = str(value if value is not None else '').strip().replace(',', '')
    if not re.fullmatch(r'\d{1,6}(?:\.0+)?', s):
        return None
    n = int(float(s))
    return n if n <= 200000 else None


def parse_player_stats(payload, player_id):
    entries = payload.get('playerstats') if isinstance(payload, dict) else None
    if not isinstance(entries, list):
        entries = []
    groups = {}
    valid_records = 0
    for item in entries[:10]:
        if not isinstance(item, dict):
            continue
        if str(item.get('idPlayer') or '') != str(player_id):
            continue
        if normalize(item.get('strSport')) not in ('soccer', 'football'):
            continue
        category = stat_category(item.get('strStatistic'))
        val = valid_stat(item.get('strValue'))
        if category is None or val is None:
            continue
        season = str(item.get('strSeason') or '').strip()
        if not re.fullmatch(r'(?:19|20)\d{2}(?:[-/](?:\d{2}|(?:19|20)\d{2}))?', season):
            continue
        league = str(item.get('strLeague') or '')[:100].strip()
        team = str(item.get('strTeam') or '')[:100].strip()
        key = (season, league, team)
        group = groups.setdefault(key, {
            'season': season, 'competition': league, 'team': team,
            'appearances': None, 'goals': None, 'assists': None,
            'minutes': None, 'yellowCards': None, 'redCards': None,
        })
        # Duplicate category rows may represent different phases; don't add
        # them up or pick an arbitrary row as an authoritative total.
        if group[category] is None:
            group[category] = val
        elif group[category] != val:
            group[category] = None
        valid_records += 1
    rows = sorted(groups.values(), key=lambda x: (x['season'], x['competition'], x['team']), reverse=True)
    return {
        'items': rows,
        'available': bool(rows),
        'partial': len(entries) >= 10,
        'recordsSeen': min(len(entries), 10),
        'source': 'TheSportsDB V1 Free (məhdud)',
        'sourceUrl': ROOT + 'lookupplayerstats.php?id=' + str(player_id),
        'scope': 'Mövsüm / yarış / klub üzrə; tam karyera deyil',
        'notice': 'Pulsuz API ən çox 10 statistika qeydi qaytarır. Göstərilməyən assist və matçlar sıfır deyil.',
    }

def openliga_matches(date):

  try:
   kick=_dt.datetime.fromisoformat(when.replace('Z','+00:00'))

   if kick.tzinfo is None:
    from zoneinfo import ZoneInfo
    kick=kick.replace(tzinfo=ZoneInfo('Europe/Berlin'))

  except (ValueError,TypeError):continue

 return out

def espn_standings(league):
 """Keep ESPN's real group/goal/form fields rather than making up table data."""
 j=source_call('ESPN',f'https://site.api.espn.com/apis/v2/sports/soccer/{league}/standings',900)
 def num(v):
  try:
   if v is None or v=='':return None
   return int(float(v))
  except (TypeError,ValueError,OverflowError):return None
 def normalize_entries(entries,group=''):
  result=[]
  for index,r in enumerate(entries if isinstance(entries,list) else []):
   t=r.get('team') or {};stats=r.get('stats') or []
   def raw(*names):
    found=next((v for v in stats if v.get('name') in names or v.get('abbreviation') in names),None)
    return (found.get('value') if found.get('value') is not None else found.get('displayValue')) if found else None
   logos=t.get('logos') or []
   gf=num(raw('pointsFor','goalsFor','goalsScored','GF'))
   ga=num(raw('pointsAgainst','goalsAgainst','goalsConceded','GA'))
   gd=num(raw('pointDifferential','goalDifference','GD'))
   # Form is only displayed when explicitly reported as W/D/L sequence.
   form=str(raw('lastFive','form','last5','lastFiveGames') or '').upper().replace(' ','')
   if not form or len(form)>10 or any(ch not in 'WDL-' for ch in form):form=''
   row={'name':t.get('displayName') or t.get('name') or '',
        'image':logos[0].get('href','') if logos else t.get('logo',''),
        'played':num(raw('gamesPlayed','GP')),'win':num(raw('wins','W')),
        'draw':num(raw('ties','D')),'loss':num(raw('losses','L')),
        'points':num(raw('points','PTS')),'rank':num(raw('rank','RK')) or index+1,
        'gf':gf,'ga':ga,'gd':gd,'form':form,'group':str(group or '')[:80]}
   if row['name']:result.append(row)
  return result
 out=normalize_entries((j.get('standings') or {}).get('entries'))
 def walk(nodes,depth=0):
  if depth>4 or not isinstance(nodes,list):return []
  found=[]
  for g in nodes:
   if not isinstance(g,dict):continue
   name=g.get('name') or g.get('abbreviation') or g.get('shortName') or ''
   found.extend(normalize_entries((g.get('standings') or {}).get('entries'),name))
   found.extend(walk(g.get('children'),depth+1))
  return found
 out.extend(walk(j.get('children')))
 # Duplicate teams may appear at parent and group level; prefer explicit group.
 merged={}
 for row in out:
  key=(row['group'],row['name'])
  merged[key]=row
 if any(row.get('group') for row in merged.values()):
  # Avoid parent aggregate duplicating group members.
  grouped={row['name'] for row in merged.values() if row['group']}
  merged={k:v for k,v in merged.items() if v['group'] or v['name'] not in grouped}
 return list(merged.values())

def openliga_table():
 season=time.gmtime().tm_year-(time.gmtime().tm_mon<7)
 rows=get_json(f'https://api.openligadb.de/getmatchdata/bl1/{season}',ttl=900)
 teams={}
 for m in rows:
  if not m.get('matchIsFinished'):continue
  results=m.get('matchResults') or []
  final=next((x for x in results if x.get('resultTypeID')==2),results[-1] if results else None)
  if final is None:continue
  a=m.get('team1',{}).get('teamName');b=m.get('team2',{}).get('teamName');x=final.get('pointsTeam1');y=final.get('pointsTeam2')
  if not (a and b and isinstance(x,int) and isinstance(y,int)):continue
  for name,goals,against in [(a,x,y),(b,y,x)]:
   row=teams.setdefault(name,{'name':name,'played':0,'win':0,'draw':0,'loss':0,'points':0,'gf':0,'ga':0,'image':''})
   row['played']+=1;row['gf']+=goals;row['ga']+=against
   if goals>against:row['win']+=1;row['points']+=3
   elif goals==against:row['draw']+=1;row['points']+=1
   else:row['loss']+=1
 return [dict(row,rank=i+1) for i,row in enumerate(sorted(teams.values(),key=lambda x:(-x['points'],-(x['gf']-x['ga']),-x['gf'],x['name'])))]
