func textVal(v any) string {
	switch x := v.(type) {
	case string:
		return x
	case float64:
		return strconv.FormatFloat(x, 'f', -1, 64)
	case json.Number:
		return string(x)
	case bool:
		if x {
			return "true"
		}
		return "false"
	case nil:
		return ""
	default:
		return fmt.Sprint(x)
	}
}

func localBakuDate(d time.Time) string {
	z, err := time.LoadLocation("Asia/Baku")
	if err != nil {
		return d.UTC().Add(4 * time.Hour).Format("2006-01-02")
	}
	return d.In(z).Format("2006-01-02")

// The warm-up probe should never hold the status spinner for a full minute.
// Render Free may need ~1 minute to wake; short probes are retried while the UI stays responsive.
var client = &http.Client{Timeout: 65 * time.Second}
var statusClient = &http.Client{Timeout: 8 * time.Second}

func apiGet(path string) (map[string]any, error) {
	c := client
	if path == "/healthz" {
		c = statusClient
	}
	r, err := http.NewRequest(http.MethodGet, backend+path, nil)
	if err != nil {
		return nil, err
	}
	r.Header.Set("Accept", "application/json")
	r.Header.Set("User-Agent", "MATCHLABAZ-NativePortable/0.9.49")
	resp, err := c.Do(r)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()
	if resp.StatusCode < 200 || resp.StatusCode >= 300 {
		return nil, fmt.Errorf("server HTTP %d", resp.StatusCode)
	}
	var out map[string]any
	dec := json.NewDecoder(resp.Body)
	if err := dec.Decode(&out); err != nil {
		return nil, err
	}
	return out, nil
}
func enqueue(key string, job func() error) {
	state.Lock()
	if state.busy[key] {
		state.Unlock()
		return
	}
	state.busy[key] = true
	state.stamp[key]++
	seq := state.stamp[key]
	state.Unlock()
	uiInvalidate()
	go func() {
		err := job()
		state.Lock()
		if state.stamp[key] == seq {
			state.busy[key] = false
			if err != nil {
				state.errorText = err.Error()
				if key == "status" {
					// Don't misreport a Render cold start as a permanent failure.
					if time.Since(state.wakeBegan) < 95*time.Second {
						state.server = "waking"
					} else {
						state.server = "offline"
					}
				}
				if key == "matches" {
					state.matchesMessage = "Bağlantı alınmadı: " + err.Error()
					state.matchesAvailable = false
				}
				if key == "standings" {
					state.tableMessage = "Cədvəl alınmadı: " + err.Error()
				}
			}
		}
		state.Unlock()
		uiInvalidate()
	}()
}

// Check the smallest backend health endpoint without blocking the UI thread.
// Render Free may need a cold start; the app must never wait for it to display.
func checkStatus() {
	state.Lock()
	if state.busy["status"] {
		state.Unlock()
		return
	}
	wasOnline := state.server == "online"
	if state.wakeBegan.IsZero() {
		state.wakeBegan = time.Now()
	}
	if !wasOnline {
		age := time.Since(state.wakeBegan)
		switch {
		case age < 3500*time.Millisecond:
			state.server = "checking"
		case age < 95*time.Second:
			state.server = "waking"
		default:
			state.server = "offline"
		}
	}
	state.lastStatusAttempt = time.Now()
	state.Unlock()
	enqueue("status", func() error {
		r, e := apiGet("/healthz")
		if e != nil {
			return e
		}
		if !getBool(r, "ok") {
			return fmt.Errorf("server bazası hələ hazır deyil")
		}
		state.Lock()
		firstConnection := state.server != "online"
		state.server = "online"
		state.wakeBegan = time.Time{}
		state.checked = time.Now()
		state.errorText = ""
		if total := get(r, "registryTotal"); total != "" {
			state.worldTotal = total
		}
		state.Unlock()
		// Refresh real matches once the cold start completes. Never invent results.
		if firstConnection {
			go func() { time.Sleep(600 * time.Millisecond); loadMatches() }()
		}
		return nil
	})
}
func decodeMatches(r map[string]any) []match {
	var rows []match
	for _, raw := range getArr(r["items"]) {
		m := getObj(raw)
		home := getObj(m["home"])
		away := getObj(m["away"])
		rows = append(rows, match{ID: get(m, "id"), League: get(m, "leagueName"), Home: get(home, "name"), Away: get(away, "name"), HS: get(m, "homeScore"), AS: get(m, "awayScore"), State: get(m, "state"), Minute: get(m, "minute"), Source: get(m, "source"), Raw: m})
	}
	return rows
}
func loadMatches() {
	state.Lock()
	state.matchesMessage = "Mənbədən matç məlumatları alınır…"
	date, league := state.date, state.league
	state.Unlock()
	enqueue("matches", func() error {
		r, e := apiGet("/api/live?league=" + url.QueryEscape(leagues[league]) + "&date=" + url.QueryEscape(localBakuDate(date)))
		if e != nil {
			return e
		}
		rows := decodeMatches(r)
		state.Lock()

		state.matchesAvailable = getBool(r, "available")
		state.matchesStale = getBool(r, "stale") || getBool(r, "partial")
		state.matchesMessage = get(r, "message")
		state.Unlock()
		return nil
	})
}

func font(size int, bold bool) uintptr {
	key := size
	if bold {
		key += 1000
	}
	if v := fonts[key]; v != 0 {
		return v
	}
	w := 400
	if bold {
		w = 700
	}
	v := call(pCreateFont, uintptr(int32(-size)), 0, 0, 0, uintptr(w), 0, 0, 0, 1, 0, 0, 0, 0, ptr("Segoe UI"))
	fonts[key] = v
	return v
}
func fill(hdc uintptr, r rect, color uintptr) {
	b := call(pCreateBrush, color)
	call(pFillRect, hdc, uintptr(unsafe.Pointer(&r)), b)

}
func round(hdc uintptr, r rect, color uintptr, radius int) {
	b := call(pCreateBrush, color)
	oldB := call(pSelectObject, hdc, b)
	pen := call(pCreatePen, PS_SOLID, 0, color)
	oldP := call(pSelectObject, hdc, pen)
	call(pRoundRect, hdc, uintptr(r.L), uintptr(r.T), uintptr(r.R), uintptr(r.B), uintptr(radius), uintptr(radius))
	call(pSelectObject, hdc, oldB)
	call(pSelectObject, hdc, oldP)
	call(pDeleteObject, b)
	call(pDeleteObject, pen)
}
func ellipse(hdc uintptr, r rect, color uintptr) {
	b := call(pCreateBrush, color)
	oldB := call(pSelectObject, hdc, b)
	pen := call(pCreatePen, PS_SOLID, 0, color)
	oldP := call(pSelectObject, hdc, pen)
	call(pEllipse, hdc, uintptr(r.L), uintptr(r.T), uintptr(r.R), uintptr(r.B))
	call(pSelectObject, hdc, oldB)
	call(pSelectObject, hdc, oldP)
	call(pDeleteObject, b)
	call(pDeleteObject, pen)
}
func line(hdc uintptr, x1, y1, x2, y2 int, color uintptr, width int) {
	pen := call(pCreatePen, PS_SOLID, uintptr(width), color)
	old := call(pSelectObject, hdc, pen)
	call(pMoveToEx, hdc, uintptr(x1), uintptr(y1), 0)
	call(pLineTo, hdc, uintptr(x2), uintptr(y2))
	call(pSelectObject, hdc, old)
	call(pDeleteObject, pen)
}
func drawText(hdc uintptr, s string, r rect, sz int, color uintptr, bold bool, flags uint32) {
	call(pSetBkMode, hdc, TRANSPARENT)
	call(pSetTextColor, hdc, color)
	prev := call(pSelectObject, hdc, font(sz, bold))
	str := syscall.StringToUTF16(s)
	if len(str) > 1 {
		call(pDrawText, hdc, uintptr(unsafe.Pointer(&str[0])), uintptr(len(str)-1), uintptr(unsafe.Pointer(&r)), uintptr(flags))
	}
	call(pSelectObject, hdc, prev)
}

func pill(hdc uintptr, label string, x, y, w, h int, col uintptr) {

func panelCard(hdc uintptr, x, y, w, h int) { round(hdc, R(x, y, w, h), card, 17) }
func addHit(x, y, w, h int, action string) {
	rr := R(x, y, w, h)
	if clipContent {
		if rr.L < int32(navWidth+2) {
			rr.L = int32(navWidth + 2)
		}
		if rr.T < 76 {
			rr.T = 76
		}
		if rr.B > int32(sh) {
			rr.B = int32(sh)
		}
		if rr.R > int32(sw) {
			rr.R = int32(sw)
		}
	}
	if rr.L >= rr.R || rr.T >= rr.B {
		return
	}
	hits = append(hits, hit{rr, action})
}
func withScroll(y int) int { return y - scrolly }
func statusLabel() (string, uintptr) {
	state.Lock()
	defer state.Unlock()
	switch state.server {
	case "online":
		return "●  Server aktiv", lime
	case "offline":
		return "●  Server əlçatan deyil", red
	case "waking":
		return "●  Server oyanır", amber
	default:
		return "●  Qoşulur…", amber
	}
}

func drawSplash(hdc uintptr, w, h int) {
	fill(hdc, R(0, 0, w, h), bg)
	cx := w / 2
	cy := h/2 - 35
	elapsed := time.Since(started).Seconds()
	circle := int(55 + 8*elapsed)
	ellipse(hdc, R(cx-circle, cy-circle, circle*2, circle*2), rgb(37, 69, 55))
	ellipse(hdc, R(cx-40, cy-40, 80, 80), lime)
	drawText(hdc, "M", R(cx-36, cy-36, 72, 72), 48, bg, true, DT_CENTER|DT_VCENTER|DT_SINGLELINE)
	drawText(hdc, "MATCHLAB AZ", R(cx-350, cy+95, 700, 62), 40, ink, true, DT_CENTER|DT_VCENTER|DT_SINGLELINE)
	drawText(hdc, "FUTBOL  •  ANALİZ  •  TAKTİKA", R(cx-350, cy+157, 700, 30), 14, muted, false, DT_CENTER|DT_SINGLELINE)
	round(hdc, R(cx-135, cy+212, 270, 6), rgb(39, 60, 71), 6)
	progress := int(270 * elapsed / 1.85)
	if progress > 270 {
		progress = 270
	}
	if progress > 0 {
		round(hdc, R(cx-135, cy+212, progress, 6), lime, 6)
	}
}

	w, h := int(r.R-r.L), int(r.B-r.T)
	if w < 600 || h < 400 {
		return

	sw, sh = w, h
	hdc := screenDC
	if backDC == 0 {
		backDC = call(pCreateCompatibleDC, screenDC)
	}
	if backDC != 0 && (backBitmap == 0 || backW != w || backH != h) {
		if backBitmap != 0 {
			call(pSelectObject, backDC, backOld)
			call(pDeleteObject, backBitmap)
			backBitmap = 0
		}
		bmp := call(pCreateCompatibleBitmap, screenDC, uintptr(w), uintptr(h))
		if bmp != 0 {
			backOld = call(pSelectObject, backDC, bmp)
			backBitmap = bmp
			backW, backH = w, h

	}

func drawPitch(hdc uintptr, x, y, w, h int, formation string) {
	round(hdc, R(x, y, w, h), rgb(21, 91, 66), 18)
	for i := 0; i < 10; i++ {
		if i%2 == 0 {
			fill(hdc, R(x+15, y+i*(h-30)/10, w-30, (h-30)/10), rgb(23, 97, 70))
		}
	}
	line(hdc, x+17, y+16, x+w-17, y+16, ink, 1)
	line(hdc, x+17, y+h-16, x+w-17, y+h-16, ink, 1)

	line(hdc, x+w/2-90, y+16, x+w/2+90, y+16, ink, 1)
	line(hdc, x+w/2-90, y+h-16, x+w/2+90, y+h-16, ink, 1)

	for _, s := range strings.Split(formation, "-") {
		n, _ := strconv.Atoi(s)
		if n > 0 {
			layers = append(layers, n)
		}
	}
	num := 1
	for r, amount := range layers {
		py := y + h - 75 - r*(h-150)/(len(layers)-1)
		for i := 0; i < amount; i++ {
			px := x + (i+1)*w/(amount+1)
			ellipse(hdc, R(px-20, py-20, 40, 40), lime)
			drawText(hdc, strconv.Itoa(num), R(px-18, py-18, 36, 36), 14, bg, true, DT_CENTER|DT_VCENTER|DT_SINGLELINE)
			num++
		}
	}
}

func navigate(page string) {
	state.Lock()
	state.page = page
	state.Unlock()
	scrolly = 0
	switch page {
	case "home", "matches":
		state.Lock()
		connected := state.server == "online"
		state.Unlock()
		if connected {
			loadMatches()
		}
	case "standings":
		loadStandings()
	}
	uiInvalidate()
}
func actionDo(a string) {
	switch {
	case strings.HasPrefix(a, "nav:"):
		navigate(strings.TrimPrefix(a, "nav:"))
	case a == "quit":
		call(pDestroyWindow, hwnd)
	case a == "retry":
		state.Lock()
		if !state.busy["status"] {
			state.wakeBegan = time.Now()
		}
		state.Unlock()
		checkStatus()
	case a == "togglefullscreen":
		toggleFullscreen()
	case a == "dateprev":
		state.Lock()
		state.date = state.date.AddDate(0, 0, -1)
		state.Unlock()
		loadMatches()
		uiInvalidate()
	case a == "datenext":
		state.Lock()
		state.date = state.date.AddDate(0, 0, 1)
		state.Unlock()
		loadMatches()
		uiInvalidate()
	case a == "today":
		state.Lock()
		state.date = time.Now()
		state.Unlock()
		loadMatches()
		uiInvalidate()
	case a == "league":
		state.Lock()
		state.league = (state.league + 1) % len(leagues)
		current := state.page
		state.Unlock()
		if current == "standings" {
			loadStandings()
		} else {
			loadMatches()
		}
		uiInvalidate()
	case a == "reloadtable":
		loadStandings()
	case a == "formation":
		state.Lock()
		state.formation = (state.formation + 1) % len(formations)
		state.Unlock()
		uiInvalidate()
	case a == "dosearch":
		q := readEdit()
		if q != "" {
			loadSearch(q)
		}
		uiInvalidate()
	case strings.HasPrefix(a, "fav:"):
		id := strings.TrimPrefix(a, "fav:")
		state.Lock()
		if _, ok := state.favorites[id]; ok {
			delete(state.favorites, id)
		} else {
			for _, m := range state.matches {
				if m.ID == id {
					state.favorites[id] = m
					break
				}
			}
		}
		state.Unlock()
		saveFavorites()
		uiInvalidate()
	case strings.HasPrefix(a, "profile:"):
		id := strings.TrimPrefix(a, "profile:")
		state.Lock()
		state.page = "profile"
		state.Unlock()
		loadProfile(id)
		uiInvalidate()
	}
}
func readEdit() string {
	n := call(pGetWindowTextLength, hEdit)
	if n > 1000 {
		n = 1000
	}
	v := make([]uint16, int(n)+1)
	call(pGetWindowText, hEdit, uintptr(unsafe.Pointer(&v[0])), uintptr(len(v)))
	return strings.TrimSpace(syscall.UTF16ToString(v))
}
func toggleFullscreen() {
	style := uintptr(WS_POPUP | WS_VISIBLE)
	if fullscreen {
		fullscreen = false
		style = WS_OVERLAPPEDWINDOW | WS_VISIBLE
	} else {
		fullscreen = true
	}
	gwl := int32(GWL_STYLE)
	call(pSetWindowLongPtr, hwnd, uintptr(gwl), style)
	if fullscreen {
		w := call(pGetSystemMetrics, 0)
		h := call(pGetSystemMetrics, 1)
		call(pSetWindowPos, hwnd, 0, 0, 0, w, h, SWP_FRAMECHANGED|SWP_NOZORDER)
	} else {
		w := call(pGetSystemMetrics, 0)
		h := call(pGetSystemMetrics, 1)
		call(pSetWindowPos, hwnd, 0, (w-1220)/2, (h-820)/2, 1220, 820, SWP_FRAMECHANGED|SWP_NOZORDER)
	}
	uiInvalidate()
}
func wndProc(h uintptr, m uint32, w, l uintptr) uintptr {
	switch m {
	case WM_CREATE:
		hwnd = h
		started = time.Now()
		call(pSetTimer, h, 1, 33, 0)
		// Separate network-state housekeeping from the splash animation timer.
		call(pSetTimer, h, 2, 1500, 0)
		go func() { time.Sleep(100 * time.Millisecond); checkStatus() }()
		hEdit = call(pCreateWindowEx, WS_EX_CLIENTEDGE, ptr("EDIT"), ptr(""), WS_CHILD|WS_BORDER|WS_TABSTOP|ES_AUTOHSCROLL, 0, 0, 500, 38, h, 1001, 0, 0)
		return 0
	case WM_TIMER:
		if w == 1 {
			if time.Since(started) > 1950*time.Millisecond {
				call(pKillTimer, h, 1)
			}
			call(pInvalidateRect, h, 0, 0) // Splash animation only.
		} else if w == 2 {
			state.Lock()
			elapsed := time.Since(state.lastStatusAttempt)
			busy := state.busy["status"]
			changed := false
			if busy && state.server == "checking" && elapsed > 3500*time.Millisecond {
				state.server = "waking"
				changed = true
			}
			shouldCheck := !busy && (state.server != "online" && elapsed > 12*time.Second ||
				state.server == "online" && time.Since(state.checked) > 70*time.Second)
			state.Unlock()
			if shouldCheck {
				checkStatus()
			}
			if changed {
				call(pInvalidateRect, h, 0, 0)
			}
		}
		return 0
	case WM_ERASEBKGND:
		return 1 // Prevent Win32 from erasing the background before a buffered paint.
	case WM_PAINT:
		paint(h)
		return 0
	case WM_APP_RESULT:
		call(pInvalidateRect, h, 0, 0)
		return 0
	case WM_SIZE:
		call(pInvalidateRect, h, 0, 0)
		return 0
	case WM_CLOSE:
		call(pDestroyWindow, h)
		return 0
	case WM_DESTROY:
		if backDC != 0 {
			if backBitmap != 0 {
				call(pSelectObject, backDC, backOld)
				call(pDeleteObject, backBitmap)
				backBitmap = 0
			}
			call(pDeleteDC, backDC)
			backDC = 0
		}
		for _, f := range fonts {
			call(pDeleteObject, f)
		}
		call(pPostQuitMessage, 0)
		return 0
	case WM_KEYDOWN:
		if w == VK_ESCAPE {
			if fullscreen {
				toggleFullscreen()
			}
			return 0
		}
		if w == VK_F11 {
			toggleFullscreen()
			return 0
		}
		if w == VK_RETURN {
			state.Lock()
			page := state.page
			state.Unlock()
			if page == "search" {
				actionDo("dosearch")
			}
			return 0
		}
	case WM_LBUTTONDOWN:
		if time.Since(started) < 1900*time.Millisecond {
			return 0
		}
		x := int(int16(uint16(l)))
		y := int(int16(uint16(l >> 16)))
		for i := len(hits) - 1; i >= 0; i-- {
			if hits[i].x.contains(x, y) {
				actionDo(hits[i].action)
				return 0
			}
		}
		if y < 73 && x > navWidth && x < sw-400 {
			call(pSendMessage, h, WM_NCLBUTTONDOWN, HTCAPTION, l)
			return 0
		}
	case WM_MOUSEWHEEL:
		// Keep fractional wheel/trackpad deltas; do not lose them on each event.
		wheelRemainder += int(int16(uint16(w >> 16)))
		steps := wheelRemainder / 120
		wheelRemainder %= 120
		if steps == 0 {
			return 0
		}
		prev := scrolly
		scrolly -= steps * 54
		if scrolly < 0 {
			scrolly = 0
		}
		if scrolly > maxScroll {
			scrolly = maxScroll
		}
		if scrolly != prev {
			call(pInvalidateRect, h, 0, 0)
		}
		return 0
	case WM_COMMAND:
		if int(w&0xffff) == 1001 && int((w>>16)&0xffff) == EN_CHANGE {
			return 0
		}
	}
	return call(pDefWindowProc, h, uintptr(m), w, l)
}
func main() {
	parseCatalog()
	loadFavorites()
	fonts = map[int]uintptr{}
	call(user32.NewProc("SetProcessDPIAware"))
	className := ptr("MATCHLABNativePortableWindow")
	hInst := call(kernel32.NewProc("GetModuleHandleW"), 0)
	cursor := call(pLoadCursor, 0, IDC_ARROW)
	wc := windowClass{CbSize: uint32(unsafe.Sizeof(windowClass{})), WndProc: syscall.NewCallback(wndProc), Instance: hInst, Cursor: cursor, Background: uintptr(COLOR_WINDOW + 1), ClassName: className}
	if call(pRegisterClassEx, uintptr(unsafe.Pointer(&wc))) == 0 {
		panic("Windows registration failed")
	}
	w := call(pGetSystemMetrics, 0)
	h := call(pGetSystemMetrics, 1)
	hwnd = call(pCreateWindowEx, 0, className, ptr("MATCHLAB AZ · Futbol Mərkəzi"), WS_POPUP|WS_VISIBLE, 0, 0, w, h, 0, 0, hInst, 0)
	if hwnd == 0 {
		panic("Window creation failed")
	}
	setIcon(hwnd)
	call(pShowWindow, hwnd, SW_SHOW)
	call(pUpdateWindow, hwnd)
	var msg msgStruct
	for {
		v := call(pGetMessage, uintptr(unsafe.Pointer(&msg)), 0, 0, 0)
		if v == 0 || v == ^uintptr(0) {
			break
		}
		call(pTranslateMessage, uintptr(unsafe.Pointer(&msg)))
		call(pDispatchMessage, uintptr(unsafe.Pointer(&msg)))
	}
}
func setIcon(h uintptr) {
	data, e := bundled.ReadFile("icon.ico")
	if e != nil {
		return
	}
	p := filepath.Join(os.TempDir(), "matchlab_native_0950.ico")
	if os.WriteFile(p, data, 0600) != nil {
		return
	}
	icon := call(pLoadImage, 0, ptr(p), IMAGE_ICON, 0, 0, LR_LOADFROMFILE|LR_DEFAULTSIZE)
	if icon != 0 {
		call(pSendMessage, h, WM_SETICON, 0, icon)
		call(pSendMessage, h, WM_SETICON, 1, icon)
	}
}

//go:build windows

// MATCHLAB AZ Windows user-level setup. Native Go, no CMD window, no UAC.
package main

import (
	"embed"
	"fmt"
	"io"
	"os"
	"os/exec"
	"path/filepath"
	"strings"
	"syscall"
	"time"
	"unsafe"
)

//go:embed payload/MATCHLAB_AZ.exe payload/icon.ico
var payload embed.FS

const (
	appName     = "MATCHLAB AZ"
	version     = "0.9.50"
	mbOkCancel  = 0x00000001
	mbIconInfo  = 0x00000040
	mbIconError = 0x00000010
)

var user32 = syscall.NewLazyDLL("user32.dll")
var messageBoxW = user32.NewProc("MessageBoxW")

func dialog(caption, message string, flags uintptr) uintptr {
	c := syscall.StringToUTF16Ptr(caption)
	m := syscall.StringToUTF16Ptr(message)
	result, _, _ := messageBoxW.Call(0, uintptr(unsafe.Pointer(m)), uintptr(unsafe.Pointer(c)), flags)
	return result
}
func quotePS(s string) string { return "'" + strings.ReplaceAll(s, "'", "''") + "'" }
func execPS(script string) error {
	cmd := exec.Command("powershell.exe", "-NoLogo", "-NoProfile", "-NonInteractive", "-WindowStyle", "Hidden", "-Command", script)
	cmd.SysProcAttr = &syscall.SysProcAttr{HideWindow: true}
	data, err := cmd.CombinedOutput()
	if err != nil {
		return fmt.Errorf("Windows shortcut / application registration: %w: %s", err, strings.TrimSpace(string(data)))
	}
	return nil
}
func installDir() (string, error) {
	local := os.Getenv("LOCALAPPDATA")
	if local == "" {
		home, err := os.UserHomeDir()
		if err != nil {
			return "", err
		}
		local = filepath.Join(home, "AppData", "Local")
	}
	return filepath.Join(local, "Programs", "MATCHLAB AZ"), nil
}
func writePayload(name, target string) error {
	source, err := payload.Open(name)
	if err != nil {
		return err
	}
	defer source.Close()
	tmp := target + ".new"
	dest, err := os.OpenFile(tmp, os.O_WRONLY|os.O_CREATE|os.O_TRUNC, 0600)
	if err != nil {
		return err
	}
	_, copyErr := io.Copy(dest, source)
	closeErr := dest.Close()
	if copyErr != nil {
		return copyErr
	}
	if closeErr != nil {
		return closeErr
	}
	// Removing an active executable fails on Windows. Report this rather than corrupting it.
	if err := os.Remove(target); err != nil && !os.IsNotExist(err) {
		return fmt.Errorf("Yeniləmə üçün açıq MATCHLAB AZ proqramını bağla: %w", err)
	}
	if err := os.Rename(tmp, target); err != nil {
		return err
	}
	return nil
}
func install() error {
	dir, err := installDir()
	if err != nil {
		return err
	}
	if err := os.MkdirAll(dir, 0700); err != nil {
		return err
	}
	exe := filepath.Join(dir, "MATCHLAB_AZ.exe")
	icon := filepath.Join(dir, "matchlab_icon.ico")
	if err := writePayload("payload/MATCHLAB_AZ.exe", exe); err != nil {
		return err
	}
	if err := writePayload("payload/icon.ico", icon); err != nil {
		return err
	}
	self, err := os.Executable()
	if err != nil {
		return err
	}
	uninstall := filepath.Join(dir, "MATCHLAB_Uninstall.exe")
	// A dedicated uninstaller is registered in Settings > Apps > Installed apps.
	if !strings.EqualFold(self, uninstall) {
		in, e := os.Open(self)
		if e != nil {
			return e
		}
		defer in.Close()
		dst, e := os.Create(uninstall)
		if e != nil {
			return e
		}
		_, e = io.Copy(dst, in)
		closeErr := dst.Close()
		if e != nil {
			return e
		}
		if closeErr != nil {
			return closeErr
		}
	}
	ps := "$ErrorActionPreference='Stop';" +
		"$ws=New-Object -ComObject WScript.Shell;" +
		"$desktop=[Environment]::GetFolderPath('DesktopDirectory');" +
		"$start=[Environment]::GetFolderPath('Programs');" +
		"foreach($folder in @($desktop,$start)){" +
		"if([string]::IsNullOrWhiteSpace($folder)){throw 'Windows Desktop or Start path missing'};" +
		"New-Item -ItemType Directory -Path $folder -Force | Out-Null;" +
		"$link=$ws.CreateShortcut((Join-Path $folder 'MATCHLAB AZ.lnk'));" +
		"$link.TargetPath=" + quotePS(exe) + ";" +
		"$link.WorkingDirectory=" + quotePS(dir) + ";" +
		"$link.IconLocation=" + quotePS(icon+",0") + ";" +
		"$link.Description='MATCHLAB AZ - Futbol mərkəzi';" +
		"$link.Save();" +
		"if(-not(Test-Path (Join-Path $folder 'MATCHLAB AZ.lnk'))){throw 'Shortcut not created'};" +
		"};" +
		"$reg='HKCU:\\Software\\Microsoft\\Windows\\CurrentVersion\\Uninstall\\MATCHLAB_AZ';" +
		"New-Item -Path $reg -Force | Out-Null;" +
		"$fields=@{DisplayName='MATCHLAB AZ';DisplayVersion='0.9.50';Publisher='MATCHLAB AZ';" +
		"InstallLocation=" + quotePS(dir) + ";" +
		"DisplayIcon=" + quotePS(icon) + ";" +
		"UninstallString=" + quotePS("\""+uninstall+"\" --uninstall") + ";" +
		"NoModify=1;NoRepair=1};" +
		"foreach($key in $fields.Keys){" +
		"$type=if($key -eq 'NoModify' -or $key -eq 'NoRepair'){'DWord'}else{'String'};" +
		"New-ItemProperty -Path $reg -Name $key -Value $fields[$key] -PropertyType $type -Force | Out-Null}" +
		";"
	if err := execPS(ps); err != nil {
		return err
	}
	cmd := exec.Command(exe)
	if err := cmd.Start(); err != nil {
		return fmt.Errorf("Tətbiq quraşdırıldı, amma açıla bilmədi: %w", err)
	}
	return cmd.Process.Release()
}
func uninstall() error {
	dir, err := installDir()
	if err != nil {
		return err
	}
	ps := "$ErrorActionPreference='Stop';" +
		"foreach($folder in @([Environment]::GetFolderPath('DesktopDirectory'),[Environment]::GetFolderPath('Programs'))){" +
		"if($folder){$link=Join-Path $folder 'MATCHLAB AZ.lnk';Remove-Item -LiteralPath $link -Force -ErrorAction SilentlyContinue}};" +
		"Remove-Item -LiteralPath 'HKCU:\\Software\\Microsoft\\Windows\\CurrentVersion\\Uninstall\\MATCHLAB_AZ' -Recurse -Force -ErrorAction SilentlyContinue;"
	if err := execPS(ps); err != nil {
		return err
	}
	_ = os.Remove(filepath.Join(dir, "MATCHLAB_AZ.exe"))
	_ = os.Remove(filepath.Join(dir, "matchlab_icon.ico"))
	// Windows cannot delete its own running uninstaller. Schedule clean-up after exit.
	cleanup := "Start-Sleep -Milliseconds 900; Remove-Item -LiteralPath " + quotePS(dir) + " -Recurse -Force -ErrorAction SilentlyContinue"
	cmd := exec.Command("powershell.exe", "-NoLogo", "-NoProfile", "-NonInteractive", "-WindowStyle", "Hidden", "-Command", cleanup)
	cmd.SysProcAttr = &syscall.SysProcAttr{HideWindow: true}
	if err := cmd.Start(); err == nil {
		_ = cmd.Process.Release()
	}
	return nil
}
func main() {
	if len(os.Args) > 1 && os.Args[1] == "--uninstall" {
		if dialog("MATCHLAB AZ", "MATCHLAB AZ proqramını silmək istəyirsən? Şəxsi seçilmiş matçlar saxlanacaq.", mbOkCancel|mbIconInfo) != 1 {
			return
		}
		if err := uninstall(); err != nil {
			dialog("MATCHLAB AZ", "Silmə alınmadı:\n"+err.Error(), mbIconError)
		}
		return
	}
	if len(os.Args) > 1 && os.Args[1] == "--self-test" {
		for _, name := range []string{"payload/MATCHLAB_AZ.exe", "payload/icon.ico"} {
			info, err := payload.Open(name)
			if err != nil {
				os.Exit(1)
			}
			stat, _ := info.Stat()
			_ = info.Close()
			if stat == nil || stat.Size() < 1000 {
				os.Exit(2)
			}
		}
		os.Exit(0)
	}
	if dialog("MATCHLAB AZ v"+version, "MATCHLAB AZ proqramı Windows-a quraşdırılsın?\n\n• Masaüstü qısayolu\n• Başlat menyusu qısayolu\n• Ayarlar > Yüklü tətbiqlər\n\nAdministrator icazəsi tələb olunmur.", mbOkCancel|mbIconInfo) != 1 {
		return
	}
	if err := install(); err != nil {
		dialog("MATCHLAB AZ - Quraşdırma xətası", err.Error(), mbIconError)
		return
	}
	dialog("MATCHLAB AZ", "Quraşdırma tamamlandı! Masaüstündə MATCHLAB AZ qısayolu yaradıldı.\n\nProqram avtomatik açılır.", mbIconInfo)
	time.Sleep(80 * time.Millisecond)
}

# Store MSIX package build - requires Windows 10/11 SDK (MakeAppx.exe).
# The Microsoft Store signs MSIX after successful submission/certification.
# Do not invent Publisher, PackageName: copy exact values from Partner Center -> Product identity.
param(
  [Parameter(Mandatory=$true)][string]$PackageName,
  [Parameter(Mandatory=$true)][string]$PublisherDistinguishedName,
  [Parameter(Mandatory=$true)][string]$PublisherDisplayName
)
$ErrorActionPreference = 'Stop'
if ($PackageName -match '^__' -or $PublisherDistinguishedName -notmatch '^CN=') {
  throw 'Use the exact Package Identity Name and Publisher distinguished name from Partner Center.'
}
$work = Join-Path $PSScriptRoot 'MSIX_STAGING'
New-Item -Path $work -ItemType Directory -Force | Out-Null
Copy-Item (Join-Path $PSScriptRoot 'MATCHLAB_AZ.exe') -Destination $work -Force
Copy-Item (Join-Path $PSScriptRoot 'Assets') -Destination $work -Recurse -Force
[xml]$xml = Get-Content -LiteralPath (Join-Path $PSScriptRoot 'AppxManifest.template.xml') -Raw
$ns = New-Object System.Xml.XmlNamespaceManager($xml.NameTable)
$ns.AddNamespace('m','http://schemas.microsoft.com/appx/manifest/foundation/windows10')
$identity=$xml.SelectSingleNode('/m:Package/m:Identity',$ns)
$identity.SetAttribute('Name',$PackageName)
$identity.SetAttribute('Publisher',$PublisherDistinguishedName)
$display=$xml.SelectSingleNode('/m:Package/m:Properties/m:PublisherDisplayName',$ns)
$display.InnerText=$PublisherDisplayName
$xml.Save((Join-Path $work 'AppxManifest.xml'))
$kitDir = Join-Path (${env:ProgramFiles(x86)}) 'Windows Kits\10\bin'
if (!(Test-Path $kitDir)) { throw 'Windows 10/11 SDK required. Install Windows SDK to get MakeAppx.exe.' }
$tool = Get-ChildItem -Path $kitDir -Recurse -Filter makeappx.exe -File |
  Where-Object { $_.FullName -match '\\x64\\makeappx\.exe$' } |
  Sort-Object FullName -Descending | Select-Object -First 1
if (!$tool) { throw 'Cannot find Windows SDK x64 makeappx.exe' }
$out = Join-Path $PSScriptRoot 'MATCHLAB_AZ_Store_v0949_UNSIGNED.msix'
& $tool.FullName pack /d $work /p $out /o
if ($LASTEXITCODE -ne 0 -or !(Test-Path $out)) { throw 'MSIX packaging failed; check MakeAppx errors.' }
Write-Host 'MSIX ready for Microsoft Store upload after package identity and certification checks:' $out
Write-Warning 'This is UNSIGNED and intended for Store submission; do not sideload to users.'
