function soundNotification(category='goal',preview=false){
 if(!preview&&!st.notifyConfig.sound)return false;
 const volume=Math.max(0,Math.min(100,Number(st.notifyConfig.volume??55)))/100;
 if(volume===0)return false;
 const pattern=NOTIFY_SOUND_PATTERNS[category];if(!pattern)return false;
 try{
  const AC=window.AudioContext||window.webkitAudioContext;if(!AC)return false;
  if(!notifyAudioContext||notifyAudioContext.state==='closed')notifyAudioContext=new AC();
  const context=notifyAudioContext;
  const play=()=>{
   if(context.state!=='running')return;
   const when=context.currentTime+.02,master=.14*volume;
   for(const [delay,hz,seconds,wave,loudness,glide] of pattern){
    const osc=context.createOscillator(),gain=context.createGain(),begin=when+delay,end=begin+seconds;
    osc.type=wave;osc.frequency.setValueAtTime(hz,begin);
    if(glide)osc.frequency.exponentialRampToValueAtTime(glide,end);
    gain.gain.setValueAtTime(.0001,begin);
    gain.gain.linearRampToValueAtTime(Math.max(.0002,master*loudness),begin+Math.min(.025,seconds*.25));
    gain.gain.setValueAtTime(Math.max(.0002,master*loudness),begin+seconds*.60);
    gain.gain.exponentialRampToValueAtTime(.0001,end);
    osc.connect(gain);gain.connect(context.destination);
    osc.start(begin);osc.stop(end+.003);
    osc.onended=()=>{try{osc.disconnect();gain.disconnect()}catch{}};
   }
  };
  if(context.state==='suspended')context.resume().then(play).catch(()=>{if(preview)toast('Brauzer səsə icazə vermədi')});
  else play();
  return true;
 }catch{if(preview)toast('Brauzer bu səsi ifa edə bilmədi');return false}
}

  if (location.protocol !== 'https:' && !['localhost','127.0.0.1'].includes(location.hostname)) return;

  if (!installButton) return;
  let pendingPrompt = null;
  const installed = () => window.matchMedia('(display-mode: standalone)').matches
    || Boolean(window.navigator.standalone);
  function hide() { installButton.hidden = true; pendingPrompt = null; }
  if (installed()) hide();
  window.addEventListener('beforeinstallprompt', e => {
    if (installed()) return;
    e.preventDefault(); pendingPrompt = e; installButton.hidden = false;
  });
  installButton.addEventListener('click', async () => {
    if (!pendingPrompt) return;
    const prompt = pendingPrompt;
    pendingPrompt = null;
    installButton.hidden = true;
    try { await prompt.prompt(); await prompt.userChoice; }
    catch { /* Browser may reject or cancel without damaging the app. */ }
  });
  window.addEventListener('appinstalled', hide);
  if ('serviceWorker' in navigator)
    window.addEventListener('load', () => {
      navigator.serviceWorker.register('/service-worker.js', {scope:'/'})
        .catch(error => console.warn('MATCHLAB AZ offline-shell registration failed:',error));
    }, {once:true});
})();

class Handler(http.server.SimpleHTTPRequestHandler):
    def __init__(self,*args,**kwargs):
        super().__init__(*args,directory=str(BASE),**kwargs)
    def guess_type(self,path):
        if path.endswith('.webmanifest'):return 'application/manifest+json'
        return super().guess_type(path)
    def do_POST(self):
        self.send_error(403,'Read-only local preview')
    def do_GET(self):
        route=urllib.parse.urlsplit(self.path)
        if route.path.startswith('/api/'):
            if len(self.path)>1200:
                self.send_error(414);return
            try:
                req=urllib.request.Request(UPSTREAM+self.path,headers={'Accept':'application/json','User-Agent':'MATCHLAB-PWA-Local/0.9.42'})
                with urllib.request.urlopen(req,timeout=45) as r:
                    data=r.read(8000001)
                    if len(data)>8000000:raise ValueError('Response too large')
                    self.send_response(r.status)
                    self.send_header('Content-Type','application/json; charset=utf-8')
                    self.send_header('Cache-Control','no-store')
                    self.send_header('Content-Length',str(len(data)))
                    self.end_headers();self.wfile.write(data)
            except urllib.error.HTTPError as e:
                self.send_error(e.code,'Backend temporarily unavailable')
            except Exception:
                data=b'{"available":false,"error":"Server connection unavailable"}'
                self.send_response(503)
                self.send_header('Content-Type','application/json')
                self.send_header('Content-Length',str(len(data)))
                self.end_headers();self.wfile.write(data)
            return
        name=urllib.parse.unquote(route.path or '/')
        if name not in PUBLIC:
            self.send_error(404);return
        self.path='/index.html' if name=='/' else name
        return super().do_GET()

/* MATCHLAB AZ installable PWA shell. No football/live/API data is cached. */
'use strict';
const CACHE_NAME = 'matchlab-az-shell-v0942';
const SHELL = [
  '/', '/index.html', '/styles.css', '/app.js',
  '/catalog.js', '/featured_media.js', '/coach_tactics.js',
  '/shell.css', '/shell.js', '/install.js', '/manifest.webmanifest',
  '/assets/icon.svg', '/assets/pwa-192.png', '/assets/pwa-512.png',
  '/assets/pwa-512-maskable.png'
];
self.addEventListener('install', event => {
  event.waitUntil((async () => {
    const cache = await caches.open(CACHE_NAME);
    await cache.addAll(SHELL.map(path => new Request(path, { cache: 'reload' })));
    await self.skipWaiting();
  })());
});
self.addEventListener('activate', event => {
  event.waitUntil((async () => {
    const names = await caches.keys();
    await Promise.all(names.filter(name => name.startsWith('matchlab-az-shell-') && name !== CACHE_NAME).map(name => caches.delete(name)));
    await self.clients.claim();
  })());
});
self.addEventListener('fetch', event => {
  const request = event.request;
  if (request.method !== 'GET') return;
  const url = new URL(request.url);
  if (url.origin !== self.location.origin) return;
  // Always ask the backend for live results, scores, profiles and standings.
  // Never present cached football data as current.
  if (url.pathname.startsWith('/api/')) return;
  if (request.mode === 'navigate') {
    event.respondWith((async () => {
      try { return await fetch(request); }
      catch {
        const cache = await caches.open(CACHE_NAME);
        return (await cache.match('/index.html')) ||
          new Response('Offline — MATCHLAB AZ interfeysi mövcud deyil.', { status: 503, headers: {'Content-Type':'text/plain; charset=utf-8'} });
      }
    })());
    return;
  }
  if (!SHELL.includes(url.pathname)) return;
  event.respondWith((async () => {
    const cache = await caches.open(CACHE_NAME);
    return (await cache.match(url.pathname)) || fetch(request);
  })());
});
